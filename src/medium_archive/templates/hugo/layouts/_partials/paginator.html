{{- /* Paginator for home (all posts), sections and terms (newest first), used by both the head in baseof.html and the list templates; Hugo caches a page's first paginator, so both get the same one. Returns "" for a page that does not paginate. */ -}} {{- $pager := "" -}} {{- if .IsHome -}} {{- $pager = .Paginate (where site.RegularPages "Type" "posts") -}} {{- else if in (slice "section" "term") .Kind -}} {{- $pager = .Paginate .Pages.ByDate.Reverse -}} {{- end -}} {{- return $pager -}}
